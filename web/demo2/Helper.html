PHẦN 2: CẤU TRÚC DRIVER C/C++ CHO STM32

Khi lập trình STM32 với cảm biến, dự án làm việc theo mô hình File .h (khai báo) + File .cpp (định nghĩa). Mục đích là tách giao diện (interface) khỏi phần thực thi (implementation).

2.1. Cấu trúc file header (.h)

File .h chứa: khai báo thư viện, hằng số, struct, nguyên mẫu hàm. Đây là "hợp đồng" giữa file .cpp với phần còn lại của chương trình.

#ifndef SHT31_H
#define SHT31_H

#include <Arduino.h>
#include <Wire.h>

#define SHT31_ADDR 0x44
#define SHT31_CMD_MEASURE 0x2C06
#define SHT31_CMD_SOFT_RESET 0x30A2

struct SHT31Data {
    float temperature;
    float humidity;
    bool valid;
};

class SHT31 {
public:
    SHT31();
    bool begin();
    bool readData(SHT31Data &outData);
    void softReset();
    void sleep();
    
private:
    uint8_t crc8(const uint8_t *data, int len);
};

#endif

2.2. Cấu trúc file implementation (.cpp)

File .cpp chứa: định nghĩa thân hàm, logic xử lý. Đây là phần "chạy thật".

#include "sht31.h"

SHT31::SHT31() {}

bool SHT31::begin() {
    Wire.begin();
    delay(50);
    Wire.beginTransmission(SHT31_ADDR);
    Wire.write(SHT31_CMD_SOFT_RESET >> 8);
    Wire.write(SHT31_CMD_SOFT_RESET & 0xFF);
    if (Wire.endTransmission() != 0) return false;
    delay(20);
    return true;
}

bool SHT31::readData(SHT31Data &outData) {
    Wire.beginTransmission(SHT31_ADDR);
    Wire.write(SHT31_CMD_MEASURE >> 8);
    Wire.write(SHT31_CMD_MEASURE & 0xFF);
    if (Wire.endTransmission() != 0) { outData.valid = false; return false; }
    
    delay(20);
    
    Wire.requestFrom(SHT31_ADDR, 6);
    if (Wire.available() != 6) { outData.valid = false; return false; }
    
    uint8_t raw[6];
    for (int i = 0; i < 6; i++) raw[i] = Wire.read();
    
    uint16_t rawTemp = (raw[0] << 8) | raw[1];
    uint16_t rawHum  = (raw[3] << 8) | raw[4];
    
    outData.temperature = -45 + 175.0 * rawTemp / 65535.0;
    outData.humidity    = 100.0 * rawHum / 65535.0;
    outData.valid = true;
    return true;
}

2.3. Cách include và sử dụng trong main.cpp

#include "sht31.h"
#include "pms7003.h"
#include "lora_ra08h.h"

SHT31 sht;
PMS7003 pms;
LoRaRA08H lora;

    Serial.begin(115200);

    if (sht.readData(env)) {

    }

    lora.send(payload);

}

PHẦN 3: CẤU TRÚC FIRMWARE STM32 (LOGIC CHẠY)

3.1. Mô hình State Machine (Máy trạng thái)

Firmware STM32 hoạt động theo mô hình State Machine, không chạy tuần tự liên tục. Đây là cách tối ưu pin tối đa:

[STOP MODE] ──(RTC wake 7p)──> [INIT] ──(30s)──> [MEASURE] ──(90s)──> [SEND LORA] ──(1s)──> [STOP MODE]
                                        (bật MOSFET)   (đọc cảm biến)   (TX 920MHz)         (tắt MOSFET)

3.2. Code mẫu với chế độ ngủ sâu

#include "stm32f1xx_hal.h"

RTC_HandleTypeDef hrtc;

void enterStopMode(uint32_t seconds) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_RESET);
    HAL_UART_DeInit(&huart1);
    
    RTC_AlarmTypeDef alarm = {0};
    alarm.AlarmTime.Seconds = seconds % 60;
    alarm.AlarmTime.Minutes = (seconds / 60) % 60;
    alarm.AlarmMask = RTC_ALARMMASK_DATEWEEKDAY | RTC_ALARMMASK_HOURS;
    HAL_RTC_SetAlarm_IT(&hrtc, &alarm, RTC_FORMAT_BIN);
    
    HAL_SuspendTick();
    HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);
    
    SystemClock_Config();
    HAL_ResumeTick();
    HAL_UART_Init(&huart1);
}

3.3. Thuật toán lọc nhiễu cảm biến (Sensor Filtering)

Cảm biến bụi mịn (PMS7003) và cảm biến khí thường bị nhảy số do môi trường. STM32 cần chạy thuật toán Trung bình trượt (Moving Average) trước khi gửi để biểu đồ không bị răng cưa và báo động giả.

#define FILTER_SIZE 10
float pm25_buffer[FILTER_SIZE] = {0};
int buffer_index = 0;

float filterPM25(float new_val) {
    pm25_buffer[buffer_index] = new_val;
    buffer_index = (buffer_index + 1) % FILTER_SIZE;
    
    float sum = 0;
    for (int i = 0; i < FILTER_SIZE; i++) {
        sum += pm25_buffer[i];
    }
    return sum / FILTER_SIZE;
}

5.1. Công thức nội suy tuyến tính

5.3. Code Node-RED tính AQI

const PM25_BP = [[0,25,0,50],[25,50,51,100],[50,80,101,150],[80,150,151,200],[150,250,201,300],[250,500,301,500]];

function calcSubIndex(c, bp) {
    if (c <= bp[0][0]) return bp[0][2];
    for (let i = 0; i < bp.length; i++) {
        const [cL, cH, iL, iH] = bp[i];
        if (c >= cL && c <= cH) {
            if (cH === cL) return iH;
            return Math.round((iH - iL) / (cH - cL) * (c - cL) + iL);
        }
    }
    return 500; 
}
function calcAQI(pm25) {
    return calcSubIndex(pm25, PM25_BP);
}

PHẦN 6: CƠ CHẾ ĐỒNG BỘ THỜI GIAN (RTC & NTP)

Tại Node (STM32): Sử dụng module DS3231 (RTC). Chip này có pin riêng, chạy như đồng hồ đeo tay, đếm thời gian liên tục kể cả khi STM32 ngủ. Khi mất điện, DS3231 vẫn chạy.

Tại Gateway (ESP32): ESP32 có WiFi nên dùng giao thức NTP để đồng bộ giờ chuẩn xác từ Server quốc tế. Gateway đính kèm mốc thời gian (timestamp) vào gói JSON gửi lên Server.

#include <WiFi.h>
#include <PubSubClient.h>

const char* mqtt_server = "airwatch.io.vn";
WiFiClient espClient;
PubSubClient client(espClient);

void reconnect() {
    while (!client.connected()) {
        if (client.connect("ESP32_Gateway")) {
            client.subscribe("airwatch/node01/cmd");
        } else { delay(5000); }
    }
}

void setup() {
    client.setServer(mqtt_server, 1883);
}

void loop() {
    if (!client.connected()) reconnect();
    client.loop();
    String payload = "{\"t\":30.5,\"h\":80}";
    client.publish("airwatch/node01/data", payload.c_str());
    delay(420000);
}

PHẦN 11: BẢO MẬT 2FA TOTP

const passOK = bcrypt.compareSync(inputPass, user.password_hash);
const otpOK = speakeasy.totp.verify({
    secret: user.totp_secret, encoding: 'base32', token: inputOTP, window: 1
});
if (passOK && otpOK) { /* Cấp token */ }

5.3. Code Node-RED tính AQI

const PM25_BP = [[0,25,0,50],[25,50,51,100],[50,80,101,150],[80,150,151,200],[150,250,201,300],[250,500,301,500]];

function calcSubIndex(c, bp) {
    if (c <= bp[0][0]) return bp[0][2];
    for (let i = 0; i < bp.length; i++) {
        const [cL, cH, iL, iH] = bp[i];
        if (c >= cL && c <= cH) {
            if (cH === cL) return iH;
            return Math.round((iH - iL) / (cH - cL) * (c - cL) + iL);
        }
    }
    return 500; 
}
function calcAQI(pm25) {
    return calcSubIndex(pm25, PM25_BP);
}

PHẦN 6: CƠ CHẾ ĐỒNG BỘ THỜI GIAN (RTC & NTP)

Tại Node (STM32): Sử dụng module DS3231 (RTC). Chip này có pin riêng, chạy như đồng hồ đeo tay, đếm thời gian liên tục kể cả khi STM32 ngủ. Khi mất điện, DS3231 vẫn chạy.

Tại Gateway (ESP32): ESP32 có WiFi nên dùng giao thức NTP để đồng bộ giờ chuẩn xác từ Server quốc tế. Gateway đính kèm mốc thời gian (timestamp) vào gói JSON gửi lên Server.

PHẦN 11: BẢO MẬT 2FA TOTP

});
if (passOK && otpOK) { /* Cấp token */ }

PHẦN 13: THIẾT KẾ CƠ KHÍ VÀ ĐỐI LƯU KHÍ

13.1. Kết cấu vỏ hộp

Vỏ hộp được chế tạo bằng nhựa ABS phủ lớp chống tia UV. Hộp được chia làm 3 khoang độc lập: Khoang Logic (chứa MCU, mạch nguồn), Khoang Sensor (chứa các cảm biến môi trường) và Khoang Bảo trì (chứa cổng nạp ST-Link).

13.2. Cơ chế đối lưu khí

Hệ thống sử dụng 3 quạt độc lập (0.13A/quạt). Quạt khoang Sensor hút khí vào trong 30 giây đầu mỗi chu kỳ. Hai quạt khoang Logic được điều khiển dựa trên nhiệt độ đọc từ cảm biến nhiệt trở, tự động bật khi nhiệt độ vượt 30°C.

13.3. Thiết kế chắn bức xạ nhiệt (Radiation Shield)

Để tránh sai số do ánh nắng mặt trời chiếu trực tiếp vào cảm biến nhiệt độ/độ ẩm, hệ thống sử dụng kết cấu tấm chắn xếp lớp (louvered plates). Thiết kế này cho phép không khí lưu thông nhưng chặn hoàn toàn bức xạ nhiệt.

13.4. Chống nhiễu LoRa

Dây ăng-ten sử dụng cáp đồng trục tiêu chuẩn 50Ω. Kẹp thêm lõi ferrite tại các đầu nối để triệt tiêu nhiễu tần số cao lan truyền theo dây dẫn.

PHẦN 15: CƠ CHẾ AN TOÀN VẬT LÝ

15.1. Phát hiện xâm nhập

Hệ thống trang bị bàn phím ma trận chống nước (có nắp đậy silicon) kết hợp với khóa điện từ (Solenoid lock - 5V, 0.2A). Hệ thống sử dụng 2 cấp độ mã khóa: Cấp 1 mở khoang Logic/Sensor, Cấp 2 mở khoang Bảo trì.

15.2. Cảnh báo khẩn cấp (Emergency)

Node tự động gửi cảnh báo khẩn cấp về Server trong các trường hợp: Phát hiện node bị mở hoặc phá hoại trái phép; Dung lượng pin xuống dưới ngưỡng an toàn; Node bị rơi hoặc tác động ngoại lực (phát hiện qua cảm biến gia tốc LSM6DS3).

15.3. Tự phục hồi khi gặp lỗi

STM32 hoặc cảm biến gặp lỗi bất thường trong quá trình vận hành sẽ tự động reset hệ thống, đồng thời gửi cảnh báo (warning) về server để ghi nhận sự cố.

PHẦN 17: QUY TRÌNH NẠP FIRMWARE KIT RA-08H

17.1. Chuẩn bị phần cứng

Sử dụng cáp Micro-USB kết nối Kit Ra-08H với máy tính. Máy tính sẽ tự động nhận diện cổng COM thông qua chip CH340 tích hợp trên board.

17.2. Nạp Firmware qua UART

Để vào chế độ nạp (Boot Mode), thực hiện tuần tự: Giữ nút BOOT -> Nhấn nút RST -> Thả nút RST -> Thả nút BOOT. Sau đó sử dụng tool tremo_loader.py để nạp file .bin.

python3 tremo_loader.py -p /dev/ttyUSB0 -b 115200 flash 0x08000000 "firmware.bin"
